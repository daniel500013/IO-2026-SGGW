190 private void AddCsFiles(string[] paths)
191 {
192 int added = 0, skipped = 0;
193 foreach (var path in paths)
194 {
195 try
196 {
197 // T1-11: odrzuć katalogi - folder "Testy.CS" też kończy się na ".cs"
198 if (Directory.Exists(path)) { skipped++; continue; }
199 if (!File.Exists(path)) { skipped++; continue; }
200 if (!path.EndsWith(".cs", StringComparison.OrdinalIgnoreCase)) { skipped++; continue; }
201 if (studentSolutions.Exists(s => s.FilePath == path)) continue;
202 studentSolutions.Add(new StudentSolution
203 {
204 StudentId = Path.GetFileNameWithoutExtension(path),
205 FilePath = path,
206 SourceCode = File.ReadAllText(path),
207 LastModified = File.GetLastWriteTime(path)
208 });
209 added++;
210 }
211 catch (Exception ex)
212 {
213 // T1-15: błąd jednego pliku NIE może przerwać całej operacji multi-drop
214 skipped++;
215 DebugLog($"Pominięto plik '{path}': {ex.GetType().Name}: {ex.Message}");
216 }
217 }
218 UpdateStatusBar();
219 if (skipped > 0)
220 MessageBox.Show($"Dodano {added} plik(ów). Pominięto {skipped} (foldery / pliki zablokowane / nie.cs).",
221 "Dodawanie plików", MessageBoxButtons.OK, MessageBoxIcon.Information);
222 }

474 {
