46 public Task RunAsync(List<StudentSolution> students, AnswerKey key, int timeoutMs,
47 BindingList<ResultRow> results, IProgress<int> progress)
48 {
49 var ui = SynchronizationContext.Current ?? new SynchronizationContext();
50
51 return Task.Run(() =>
52 {
53 int total = students.Count * SumTestCases(key);
54 int done = 0;
55
56 foreach (var student in students)
57 {
58 var comp = compiler.Compile(student.SourceCode);
59
60 if (!comp.Success)
61 {
62 EmitCompilationFailureForAllTasks(student, key, comp.ErrorMessage, ui, results);
63 done += SumTestCases(key);
64 progress?.Report(done * 100 / Math.Max(1, total));
65 continue;
66 }
67
68 foreach (var sheet in key.Tasks)
69 {
70 var method = runner.FindMethod(comp.Assembly, sheet.Name);
71
72 foreach (var tc in sheet.TestCases)
73 {
74 var row = (method == null)
75 ? MakeRow(student, sheet, tc, RunStatus.BrakMetody, 0, "Brak metody " + sheet.Name.Replace(" ", ""))
76 : GradeOne(student, sheet, tc, method, timeoutMs);
77
78 ui.Post(_ => results.Add(row), null);
79 done++;
80 progress?.Report(done * 100 / Math.Max(1, total));
81 }
82 }
83 }
84 });
85 }

100 private ResultRow GradeOne(StudentSolution student, TaskSheet sheet, TestCase tc, MethodInfo method, int timeoutMs)
101 {
102 object[] args;
103 try
104 {
105 args = runner.ParseArgs(tc.ParametersRaw, method.GetParameters());
106 }
107 catch (Exception ex)
108 {
109 return MakeRow(student, sheet, tc, RunStatus.ZlyFormatParametrow, 0, ex.Message);
110 }
111
112 var run = runner.InvokeWithTimeout(method, args, timeoutMs);
113
114 if (run.Status == RunStatus.Timeout)
115 return MakeRow(student, sheet, tc, RunStatus.Timeout, 0, "Przekroczono limit czasu");
116
117 if (run.Status == RunStatus.Wyjatek)
118 return MakeRow(student, sheet, tc, RunStatus.Wyjatek, 0, run.ErrorMessage);
119
120 bool correct = runner.IsCorrect(run.Value, tc.ExpectedRaw, method.ReturnType);
121
122 return correct
123 ? MakeRow(student, sheet, tc, RunStatus.Ok, 1, FormatValue(run.Value))
124 : MakeRow(student, sheet, tc, RunStatus.Bledny, 0, FormatValue(run.Value));
125 }
