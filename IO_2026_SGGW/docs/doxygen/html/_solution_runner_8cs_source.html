107 toRet[i] = arr;
108
109 }
110 else
111 {
112 toRet[i] = Convert.ChangeType(token, type, CultureInfo.InvariantCulture);
113 }
114
115 }

117

207 public bool IsCorrect(object actual, string expectedFromXlsx, Type returnType)
208 {
209 if (actual == null) return string.IsNullOrWhiteSpace(expectedFromXlsx);
210
211 if (actual is Array actualArr)
212 {
213 var expectedParts = PrepareXlsxInput(expectedFromXlsx);
214 if (actualArr.Length != expectedParts.Length) return false;
215
216
217 var elemType = returnType.GetElementType();
218 for (int i = 0; i < actualArr.Length; i++)
219 {
220 var expEl = Convert.ChangeType(expectedParts[i], elemType, CultureInfo.InvariantCulture);
221 if (elemType == typeof(double) || elemType == typeof(float))
222 {
223 if (Math.Abs(Convert.ToDouble(actualArr.GetValue(i)) - Convert.ToDouble(expEl)) >= 1e-6) return false;
224 }
225 else if (!actualArr.GetValue(i).Equals(expEl)) return false;
226 }
227 return true;
228 }
229
230
231 if (returnType == typeof(double) || returnType == typeof(float))
232 {
233 if (double.TryParse(expectedFromXlsx, NumberStyles.Any, CultureInfo.InvariantCulture, out var exp))
234 {
235 return Math.Abs(Convert.ToDouble(actual) - exp) < 1e-6;
236 }
237 }
238
239
240 return actual.ToString().Trim().Equals(expectedFromXlsx.Trim(), StringComparison.OrdinalIgnoreCase);
241 }

279 private static List<string> SplitRespectingBrackets(string s)
280 // MultiD nie jest wspierane
281 {
282 var toRet = new List<string>();
283 var sb = new StringBuilder();
284 int depth = 0;
285
286 foreach (char c in s)
287 {
288 if (c == '[') depth++;
289 else if (c == ']') depth--;
290 if (depth > 1) throw new NotImplementedException("Tablice wielowymiarowe nie są obecnie wspierane.");
291
292 if (c == ',' && depth == 0)
293 {
294 toRet.Add(sb.ToString());
295 sb.Clear();
296 }
297 else
298 {
299 sb.Append(c);
300 }
301 }
302
303 if (sb.Length > 0) { toRet.Add(sb.ToString()); }
304 return toRet;
305 }
